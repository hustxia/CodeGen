<template>
  <div class="app-container">
    <el-card shadow="never">
      <!-- 查询区 -->
      <el-form :inline="true" :model="query" size="small" @submit.prevent>
        <el-form-item label="商户单号">
          <el-input v-model="query.outTradeNo" placeholder="精确匹配" clearable style="width:200px" />
        </el-form-item>
        <el-form-item label="状态">
          <el-select v-model="query.payStatus" placeholder="全部" clearable style="width:120px">
            <el-option label="待支付" :value="0" />
            <el-option label="成功" :value="1" />
            <el-option label="已关闭" :value="2" />
            <el-option label="已退款" :value="3" />
          </el-select>
        </el-form-item>
        <el-form-item>
          <el-button type="primary" icon="Search" @click="load(1)">查询</el-button>
          <el-button icon="Refresh" @click="reset">重置</el-button>
        </el-form-item>
      </el-form>

      <!-- 列表 -->
      <el-table v-loading="loading" :data="rows" border size="small">
        <el-table-column prop="outTradeNo" label="商户单号" width="220" show-overflow-tooltip />
        <el-table-column prop="subject" label="标题" min-width="160" show-overflow-tooltip />
        <el-table-column label="金额（元）" width="110" align="right">
          <template #default="{ row }">{{ fenToYuan(row.amount) }}</template>
        </el-table-column>
        <el-table-column prop="payChannel" label="渠道" width="90" />
        <el-table-column label="状态" width="90" align="center">
          <template #default="{ row }">
            <el-tag size="small" :type="statusTagType(row.payStatus)">{{ statusText(row.payStatus) }}</el-tag>
          </template>
        </el-table-column>
        <el-table-column label="业务" width="160" show-overflow-tooltip>
          <template #default="{ row }">{{ row.businessType ? row.businessType + ' #' + row.businessId : '—' }}</template>
        </el-table-column>
        <el-table-column prop="paidAt" label="支付时间" width="160" />
        <el-table-column label="操作" width="250" fixed="right">
          <template #default="{ row }">
            <el-button type="text" size="small" @click="doQuery(row)">查单</el-button>
            <el-button v-if="row.payStatus === 0" type="text" size="small" class="danger-link" @click="doClose(row)">关单</el-button>
            <el-button v-if="row.payStatus === 0" type="text" size="small" @click="goPay(row)">去支付</el-button>
            <el-button v-if="row.payStatus === 1" type="text" size="small" class="danger-link" @click="openRefund(row)">退款</el-button>
          </template>
        </el-table-column>
      </el-table>

      <el-pagination
        style="margin-top:12px;text-align:right"
        background
        layout="total, prev, pager, next, sizes"
        :total="total"
        v-model:current-page="query.pageNum"
        v-model:page-size="query.pageSize"
        :page-sizes="[10, 20, 50]"
        @current-change="load()"
        @size-change="load(1)"
      />

      <!-- 退款弹窗：金额默认剩余可退全额（支持部分退款），确认后调渠道退款并刷新列表 -->
      <el-dialog v-model="refundDlg.visible" :title="refundDlg.title" width="420px" append-to-body>
        <div v-if="refundDlg.maxYuan > 0" class="tip" style="margin-bottom:12px">
          已退 {{ fenToYuan(refundDlg.refundedFen) }} 元，剩余可退 {{ refundDlg.maxYuan.toFixed(2) }} 元
        </div>
        <el-form label-width="110px">
          <el-form-item label="退款金额（元）">
            <el-input-number v-model="refundDlg.amountYuan" :min="0.01" :max="refundDlg.maxYuan"
              :precision="2" :step="1" style="width:200px" />
          </el-form-item>
          <el-form-item label="退款原因">
            <el-input v-model="refundDlg.reason" maxlength="100" placeholder="选填，入退款审计" />
          </el-form-item>
        </el-form>
        <template #footer>
          <el-button @click="refundDlg.visible = false">取消</el-button>
          <el-button type="primary" :loading="refundDlg.loading" @click="doRefund">确认退款</el-button>
        </template>
      </el-dialog>
    </el-card>
  </div>
</template>

<script>
import { payOrderPage, getPayOrder, closePayOrder, refundPayOrder } from '@/api/pay'

export default {
  name: 'PayOrderPage',
  data() {
    return {
      query: { pageNum: 1, pageSize: 10, outTradeNo: '', payStatus: undefined },
      rows: [],
      total: 0,
      loading: false,
      refundDlg: { visible: false, title: '', outTradeNo: '', refundedFen: 0, maxYuan: 0, amountYuan: null, reason: '', loading: false }
    }
  },
  mounted() { this.load(1) },
  methods: {
    async load(page) {
      if (page) this.query.pageNum = page
      this.loading = true
      try {
        const data = (await payOrderPage(this.query)).data
        this.rows = data.records || []
        this.total = Number(data.total || 0)
      } finally {
        this.loading = false
      }
    },
    reset() {
      this.query.outTradeNo = ''
      this.query.payStatus = undefined
      this.load(1)
    },
    async doQuery(row) {
      const data = (await getPayOrder(row.outTradeNo)).data
      Object.assign(row, data)
      this.$message.success('已同步渠道最新状态：' + this.statusText(data.payStatus))
    },
    doClose(row) {
      this.$confirm('关闭后该支付单不可再支付，确认？', '关单', { type: 'warning' })
        .then(async() => {
          await closePayOrder(row.outTradeNo)
          this.$message.success('已关闭')
          this.load()
        }).catch(() => {})
    },
    goPay(row) {
      this.$router.push({ path: '/pay/cashier', query: {
        subject: row.subject, amount: this.fenToYuan(row.amount), redirect: '/pay/order'
      } })
    },
    // V6.3 补全：退款入口——payStatus=1（含部分退款未退满）可再退剩余金额
    openRefund(row) {
      const refundedFen = Number(row.refundAmount || 0)
      const maxYuan = Math.max(0, (Number(row.amount || 0) - refundedFen) / 100)
      this.refundDlg = {
        visible: true,
        title: '退款：' + (row.subject || row.outTradeNo),
        outTradeNo: row.outTradeNo,
        refundedFen: refundedFen,
        maxYuan: maxYuan,
        amountYuan: maxYuan > 0 ? Number(maxYuan.toFixed(2)) : null,
        reason: '',
        loading: false
      }
    },
    async doRefund() {
      const dlg = this.refundDlg
      if (!dlg.amountYuan || dlg.amountYuan <= 0 || dlg.amountYuan > dlg.maxYuan) {
        this.$message.warning('退款金额必须在剩余可退范围内')
        return
      }
      dlg.loading = true
      try {
        await refundPayOrder(dlg.outTradeNo, { refundAmount: dlg.amountYuan, reason: dlg.reason || undefined })
        this.$message.success('已发起退款')
        dlg.visible = false
        await this.load()
      } finally {
        dlg.loading = false
      }
    },
    fenToYuan(fen) { return (Number(fen || 0) / 100).toFixed(2) },
    statusText(s) { return { 0: '待支付', 1: '成功', 2: '已关闭', 3: '已退款' }[s] || s },
    statusTagType(s) { return { 0: 'warning', 1: 'success', 2: 'info', 3: 'danger' }[s] || 'info' }
  }
}
</script>

<style scoped>
.danger-link { color: #f56c6c; }
</style>
