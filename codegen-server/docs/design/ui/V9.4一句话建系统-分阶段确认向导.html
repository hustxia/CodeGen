CREATE DATABASE IF NOT EXISTS `asset_demo` DEFAULT CHARACTER SET utf8mb4;

CREATE TABLE IF NOT EXISTS `asset_category` (
  `id` bigint NOT NULL COMMENT '主键',
  `category_name` varchar(64) NULL COMMENT '分类名称',
  `parent_id` bigint NULL COMMENT '上级分类',
  `create_time` datetime NULL COMMENT '创建时间',
  `update_time` datetime NULL COMMENT '更新时间',
  `create_by` bigint NULL COMMENT '创建人',
  `update_by` bigint NULL COMMENT '更新人',
  `deleted` tinyint NOT NULL DEFAULT 0 COMMENT '逻辑删除',
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='设备分类';

CREATE TABLE IF NOT EXISTS `asset_info` (
  `id` bigint NOT NULL COMMENT '主键',
  `asset_code` varchar(64) NULL COMMENT '资产编码',
  `asset_name` varchar(128) NULL COMMENT '资产名称'
  -- 其余字段与审计列由同一服务端组装器生成
);
