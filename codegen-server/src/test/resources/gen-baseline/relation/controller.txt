@Operation(summary = "分页查询商品")
@PostMapping("/page")
public Result<Page<DemoGoodsVO>> page(@RequestBody DemoGoodsPageDTO dto) {
    Page<DemoGoods> page = new Page<>(dto.getPageNum(), dto.getPageSize());
    LambdaQueryWrapper<DemoGoods> wrapper = new LambdaQueryWrapper<>();
        wrapper.like(StringUtils.hasText(dto.getGoodsName()), DemoGoods::getGoodsName, dto.getGoodsName());
        wrapper.eq(dto.getStatus() != null, DemoGoods::getStatus, dto.getStatus());
        wrapper.eq(dto.getDeptId() != null, DemoGoods::getDeptId, dto.getDeptId());
        wrapper.ge(dto.getPrice() != null, DemoGoods::getPrice, dto.getPrice());
        wrapper.orderByDesc(DemoGoods::getId);
    demoGoodsService.page(page, wrapper);
    Page<DemoGoodsVO> voPage = new Page<>();
    BeanUtils.copyProperties(page, voPage);
    List<DemoGoodsVO> records = page.getRecords().stream()
            .map(e -> {
                DemoGoodsVO vo = new DemoGoodsVO();
                BeanUtils.copyProperties(e, vo);
                return vo;
            })
            .toList();
    voPage.setRecords(records);
    return Result.ok(voPage);
}

@Operation(summary = "关联下拉选项（通用，按 ref_table 远程搜索）")
@GetMapping("/options")
public Result<List<Map<String, Object>>> options(@RequestParam("refTable") String refTable,
                                                @RequestParam(value = "label", required = false) String label,
                                                @RequestParam(value = "keyword", required = false) String keyword) {
    if (refTable == null || !refTable.matches("[A-Za-z0-9_]+")) {
        throw new BusinessException(400, "非法关联表名");
    }
    String labelCol = label == null || !label.matches("[A-Za-z0-9_]+") ? "name" : label;
    String base = "SELECT id, " + labelCol + " AS label FROM " + refTable;
    String like = keyword == null || keyword.isBlank() ? null : "%" + keyword.trim() + "%";
    // 无关键字是「列表页翻译 ID 为名称」，要一次拉够；有关键字是下拉搜索，20 条足够
    int limit = like == null ? 500 : 20;
    String sql = base + " WHERE deleted = 0" + (like == null ? "" : " AND " + labelCol + " LIKE ?")
            + " ORDER BY id DESC LIMIT " + limit;
    try {
        return Result.ok(like == null
                ? jdbcTemplate.queryForList(sql)
                : jdbcTemplate.queryForList(sql, like));
    } catch (Exception e) {
        // 关联表没有 deleted 列（如关联表）时降级为不带软删条件
        String fallback = base + (like == null ? "" : " WHERE " + labelCol + " LIKE ?")
                + " ORDER BY id DESC LIMIT " + limit;
        return Result.ok(like == null
                ? jdbcTemplate.queryForList(fallback)
                : jdbcTemplate.queryForList(fallback, like));
    }
}

@Operation(summary = "导出商品 Excel（按查询条件过滤；不传条件导出全部）")
@PostMapping("/export")
public ResponseEntity<byte[]> export(@RequestBody(required = false) DemoGoodsPageDTO query) throws IOException {
    // 复用列表页的筛选口径：导出的行必须和用户在页面上看到的一致，
    // 否则「筛出 20 条、导出 2000 条」会被当成 bug。
    LambdaQueryWrapper<DemoGoods> wrapper = new LambdaQueryWrapper<>();
    if (query != null) {
        wrapper.like(StringUtils.hasText(query.getGoodsName()), DemoGoods::getGoodsName, query.getGoodsName());
        wrapper.eq(query.getStatus() != null, DemoGoods::getStatus, query.getStatus());
        wrapper.eq(query.getDeptId() != null, DemoGoods::getDeptId, query.getDeptId());
        wrapper.ge(query.getPrice() != null, DemoGoods::getPrice, query.getPrice());
    }
    wrapper.orderByDesc(DemoGoods::getId);
    List<DemoGoods> rows = demoGoodsService.list(wrapper);
    try (Workbook wb = new XSSFWorkbook(); ByteArrayOutputStream bos = new ByteArrayOutputStream()) {
        Sheet sheet = wb.createSheet("商品");
        Row header = sheet.createRow(0);
        header.createCell(0).setCellValue("商品名称");
        header.createCell(1).setCellValue("状态");
        header.createCell(2).setCellValue("所属部门");
        header.createCell(3).setCellValue("价格");
        int rowIndex = 1;
        for (DemoGoods e : rows) {
            Row row = sheet.createRow(rowIndex++);
            row.createCell(0).setCellValue(cellText(e.getGoodsName()));
            // 字典列导出中文标签，与页面显示一致；null 写空串，不能落成字面量 "null"
            row.createCell(1).setCellValue(e.getStatus() == null ? "" : dictTexts.label("sys_status", String.valueOf(e.getStatus())));
            row.createCell(2).setCellValue(cellNum(e.getDeptId()));
            row.createCell(3).setCellValue(cellNum(e.getPrice()));
        }
        wb.write(bos);
        String fileName = URLEncoder.encode("商品.xlsx", StandardCharsets.UTF_8).replace("+", "%20");
        return ResponseEntity.ok()
                .header(HttpHeaders.CONTENT_DISPOSITION, "attachment; filename*=UTF-8''" + fileName)
                .contentType(MediaType.APPLICATION_OCTET_STREAM)
                .body(bos.toByteArray());
    }
}

    private static String cellText(Object value) {
        return value == null ? "" : String.valueOf(value);
    }

    private static double cellNum(Object value) {
        return value instanceof Number number ? number.doubleValue() : 0D;
    }

    private static String text(Row row, Integer col) {
        if (col == null || row == null) {
            return null;
        }
        String value = cellText(row.getCell(col)).trim();
        return value.isEmpty() ? null : value;
    }
