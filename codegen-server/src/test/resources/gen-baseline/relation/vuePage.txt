        <el-form-item label="状态">
          <el-select v-model="query.status" placeholder="请选择状态" clearable style="width: 160px">
            <el-option v-for="d in dicts.sys_status || []" :key="d.value" :label="d.label" :value="d.value" />
          </el-select>
        </el-form-item>
        <el-form-item label="所属部门">
          <el-select v-model="query.deptId" placeholder="请选择所属部门" clearable style="width: 160px">
            <el-option v-for="d in []" :key="d.value" :label="d.label" :value="d.value" />
          </el-select>
        </el-form-item>
        <el-form-item label="价格">
          <el-input-number v-model="query.price" controls-position="right" style="width: 160px" />
        </el-form-item>
        <el-form-item>
          <el-button v-hasPermi="['demo_goods:query']" type="primary" icon="Search" @click="handleSearch">查询</el-button>
          <el-button icon="Refresh" @click="handleReset">重置</el-button>
        </el-form-item>
      </el-form>
    </el-card>

    <!-- 列表 -->
    <el-card shadow="never" class="table-card">
      <template #header><div class="card-header">
        <div class="card-title">
          <span class="title">商品</span>
          <span class="subtitle">共 {{ total }} 条</span>
        </div>
        <div class="card-actions">
          <el-button v-hasPermi="['demo_goods:add']" type="primary" icon="Plus" size="small" @click="openDialog()">新增商品</el-button>
          <el-button icon="Download" size="small" @click="handleExport">导出</el-button>
        </div>
      </div></template>

      <el-table
        v-loading="loading"
        :data="list"
        class="data-table"
        stripe
        header-cell-class-name="table-head-cell"
        empty-text="暂无数据"
      >
        <el-table-column prop="id" label="ID" width="180" />
        <el-table-column prop="goodsName" label="商品名称" min-width="140" show-overflow-tooltip />
        <el-table-column prop="status" label="状态" min-width="130" show-overflow-tooltip
          :formatter="(row) => dictText('sys_status', row.status)" />
        <el-table-column prop="deptId" label="所属部门" min-width="130" show-overflow-tooltip
          :formatter="(row) => refText('deptId', row.deptId)" />
        <el-table-column prop="price" label="价格" width="110" align="right" show-overflow-tooltip />
        <el-table-column prop="updateTime" label="更新时间" width="170" align="center" />
        <el-table-column label="操作" width="278" fixed="right" align="center">
          <template #default="{ row }">
            <el-button type="text" @click="openDetail(row)">详情</el-button>
            <el-button type="text" @click="handlePrintRow(row)">打印</el-button>
            <el-button v-hasPermi="['demo_goods:edit']" type="text" @click="openDialog(row)">编辑</el-button>
            <el-button v-hasPermi="['demo_goods:remove']" type="text" class="danger-btn" @click="handleDelete(row)">删除</el-button>
          </template>
        </el-table-column>
      </el-table>

      <el-pagination
        class="pagination"
        background
        layout="total, sizes, prev, pager, next, jumper"
        :total="total"
        :page-sizes="[10, 20, 50]"
        :page-size="query.size"
        v-model:current-page="query.page"
        @size-change="handleSizeChange"
        @current-change="fetchList"
      />
    </el-card>

    <!-- 新增/编辑弹窗 -->
    <el-dialog :title="form.id ? '编辑商品' : '新增商品'" v-model="dialogVisible" width="560px" top="6vh" class="form-dialog">
      <el-form ref="form" :model="form" :rules="rules" label-width="110px">
        <el-form-item label="商品名称" prop="goodsName">
          <el-input v-model="form.goodsName" placeholder="请输入商品名称" />
        </el-form-item>
        <el-form-item label="状态" prop="status">
          <el-radio-group v-model="form.status">
            <el-radio v-for="d in dicts.sys_status || []" :key="d.value" :label="d.value">{{ d.label }}</el-radio>
          </el-radio-group>

          <el-select v-model="form.deptId" filterable remote clearable placeholder="输入关键字搜索所属部门"

            :loading="refLoading['deptId']" style="width: 100%">
            <el-option v-for="o in refOptions['deptId'] || []" :key="o.id" :label="o.label" :value="o.id" />
          </el-select>

        <el-form-item label="价格" prop="price">
          <el-input-number v-model="form.price" controls-position="right" style="width: 100%" />

        <el-button @click="dialogVisible = false">取消</el-button>
        <el-button type="primary" :loading="submitting" @click="submitForm">确定</el-button>
      </template>
    </el-dialog>

    <!-- 详情抽屉（只读）。主子表：宽抽屉 + el-tabs，tab1 主表字段，tab2..N 每张明细一张只读表；
         数据复用 /with-children 一次拉全，不逐 tab 发请求 -->

<script>
import {
  getDemoGoodsList,
  getDemoGoodsOptions,
  exportDemoGoods,
  createDemoGoods,
  updateDemoGoods,
  deleteDemoGoods
} from '@/api/goods'
// 字典兜底、唯一性查重、导入失败明细下载都直接用通用 request，不依赖宿主是否提供专用 api 模块
import request from '@/utils/request'

const defaultForm = () => ({
  id: undefined,
  goodsName: '',
  status: '',
  deptId: '',
  price: undefined,
})

export default {
  name: 'DemoGoods',
  data() {
    return {
      // <gen:begin key="vue-data">
      loading: false,
      list: [],
      total: 0,
      query: { page: 1, size: 10, goodsName: '', status: undefined, deptId: undefined, price: undefined },
      dialogVisible: false,
      submitting: false,
      detailVisible: false,
      detail: {},
      refOptions: {},
      refLoading: {},
      // 列表/详情里把 xxxId 显示成名称用的字典：{ 字段名: { id: label } }
      refLabels: {},
      dicts: { sys_status: [] },
      rules: {
      },
      form: defaultForm()
      // <gen:end key="vue-data">
    }
  },
  created() {
    this.loadDicts()
    this.loadRefLabels()
    this.fetchList()
  },
  methods: {
    // <gen:begin key="vue-methods">
    async loadDicts() {
      const codes = ['sys_status']
      await Promise.all(codes.map(async code => {
        try {
          // 直接走通用 request，不依赖宿主工程是否提供 @/api/dict；
          // 宿主没有字典接口时这里会抛错，下面 catch 把选项降级为空数组，页面照常可用
          const res = await request({ url: '/api/demo/system/dict/data/type/' + code, method: 'get' })
          this.$set(this.dicts, code, (res && res.data) || [])
        } catch (e) {
          this.$set(this.dicts, code, [])
        }
      }))
    },
    dictText(code, value) {
      const hit = (this.dicts[code] || []).find(d => String(d.value) === String(value))
      return hit ? hit.label : (value === undefined || value === null ? '' : value)
    },
    async fetchList() {
      this.loading = true
      try {
        const res = await getDemoGoodsList(this.query)
        this.list = res.data.list || []
        this.total = res.data.total || 0
      } catch (e) {
        this.list = []
        this.total = 0
      } finally {
        this.loading = false
      }
    },
    handleSearch() {
      this.query.page = 1
      this.fetchList()
    },
    handleReset() {
      this.query = { page: 1, size: this.query.size, goodsName: '', status: '', deptId: '', price: undefined }
      this.fetchList()
    },
    handleSizeChange(size) {
      this.query.size = size
      this.query.page = 1
      this.fetchList()
    },
    async searchRef(field, refTable, label, keyword) {
      this.$set(this.refLoading, field, true)
      try {
        const res = await getDemoGoodsOptions(refTable, label, keyword)
        this.$set(this.refOptions, field, res.data || [])
      } catch (e) {
        this.$set(this.refOptions, field, [])
      } finally {
        this.$set(this.refLoading, field, false)
      }
    },
    initRefs() {
      this.searchRef('deptId', 'sys_dept', 'dept_name', '')
    },
    /**
     * 列表里的 xxxId 显示成名称。
     * 关联表通常是字典/客户/产品这类小表，一次性拉成 { id: 名称 } 比逐行查省得多；
     * 拉不到就退回显示原始 ID，页面不会因为关联表没权限/没数据而白屏。
     */
    loadRefLabels() {
      this.fetchRefLabels('deptId', 'sys_dept', 'dept_name')
    },
    async fetchRefLabels(field, refTable, label) {
      try {
        const res = await getDemoGoodsOptions(refTable, label, '')
        const map = {}
        ;(res.data || []).forEach(o => { map[String(o.id)] = o.label })
        this.$set(this.refLabels, field, map)
      } catch (e) {
        this.$set(this.refLabels, field, {})
      }
    },
    /**
     * 把 xxxId 显示成名称：主表列与子表明细列都会用到，定义跟随 hasRefHelper
     * （主表无引用列但子表外键有引用时也必须存在，否则明细行渲染崩溃）。
     */
    refText(field, id) {
      if (id === null || id === undefined || id === '') {
        return ''
      }
      const map = this.refLabels[field]
      return (map && map[String(id)]) || id
    },
    openDetail(row) {
      this.detail = row || {}
      this.detailVisible = true
    },
    openDialog(row) {
      this.form = row ? { ...defaultForm(), ...row } : defaultForm()
      this.dialogVisible = true
      this.initRefs()
      this.$nextTick(() => {
        if (this.$refs.form) this.$refs.form.clearValidate()
      })
    },
    /**
     * 打印单据：跳转独立打印页（无后台导航壳，window.print 直出 A4）
     */
    handlePrintRow(row) {
      this.$router.push('/print/goods/' + row.id)
    },
    async submitForm() {
      const valid = await this.$refs.form.validate().catch(() => false)
      if (!valid) return
      this.submitting = true
      try {
        if (this.form.id) {
          await updateDemoGoods(this.form.id, this.form)
        } else {
          await createDemoGoods(this.form)
        }
        this.$message.success(this.form.id ? '更新成功' : '创建成功')
        this.dialogVisible = false
        this.fetchList()
      } catch (e) { /* 拦截器已提示 */ } finally {
        this.submitting = false
      }
    },
    handleDelete(row) {
      this.$confirm(`确定删除该商品吗？`, '提示', { type: 'warning' })
        .then(async () => {
          await deleteDemoGoods(row.id)
          this.$message.success('删除成功')
          if (this.list.length === 1 && this.query.page > 1) this.query.page--
          this.fetchList()
        })
        .catch(() => {})
    },
    async handleExport() {
      try {
        const res = await exportDemoGoods(this.query)
        const blob = new Blob([res.data], { type: 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet' })
        const url = window.URL.createObjectURL(blob)
        const a = document.createElement('a')
        a.href = url
        a.download = '商品.xlsx'
        a.click()
        window.URL.revokeObjectURL(url)
      } catch (e) { /* 拦截器已提示 */ }
    }
    // <gen:end key="vue-methods">
    // ===== 以下为手写区：自定义方法写在这里，重新生成不会被覆盖 =====
  }
}
</script>
