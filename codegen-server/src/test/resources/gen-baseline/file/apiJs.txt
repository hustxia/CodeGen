import request, { toPageData } from '@/utils/request'

// 商品资料接口：demo_goods_file（经网关 /api/demo）
// <gen:begin key="api-fns">
function toPageQuery(params) {
  const { page, size, ...query } = params || {}
  return { pageNum: page || 1, pageSize: size || 10, ...query }
}
export function getDemoGoodsFileList(params) {
  return request({
    url: '/api/demo/goodsFile/page',
    method: 'post',
    data: toPageQuery(params)
  }).then(toPageData)
}
export function exportDemoGoodsFile(params) {
  return request({
    url: '/api/demo/goodsFile/export',
    method: 'post',
    data: params || {},
    responseType: 'blob'
  })
}
export function getDemoGoodsFileDetail(id) {
  return request({
    url: `/api/demo/goodsFile/${id}`,
    method: 'get'
  })
}
export function createDemoGoodsFile(data) {
  return request({
    url: '/api/demo/goodsFile',
    method: 'post',
    data
  })
}
export function updateDemoGoodsFile(id, data) {
  return request({
    url: '/api/demo/goodsFile',
    method: 'put',
    data: { ...data, id }
  })
}
export function deleteDemoGoodsFile(id) {
  return request({
    url: `/api/demo/goodsFile/${id}`,
    method: 'delete'
  })
}
