  <div class="app-container">
    <!-- <gen:begin key="vue-template"> -->
    <!-- 页头：一眼看清「我在哪个页面、这页能干什么」 -->
    <div class="page-head">
      <div class="page-head__main">
        <h2 class="page-title">商品资料</h2>
        <p class="page-desc">维护商品资料数据，可按条件筛选、编辑与导出</p>
      </div>
      <div class="page-head__meta">
        <span class="meta-pill">{{ total }} 条记录</span>
      </div>
    </div>
    <!-- 筛选区 -->
    <el-card shadow="never" class="filter-card">
      <el-form :inline="true" :model="query" @submit.native.prevent>
        <el-form-item label="商品名称">
          <el-input v-model="query.goodsName" placeholder="商品名称" clearable @keyup.enter.native="handleSearch" />
        </el-form-item>
        <el-form-item>
          <el-button v-hasPermi="['demo_demo_goods_file:query']" type="primary" icon="el-icon-search" @click="handleSearch">查询</el-button>
          <el-button icon="el-icon-refresh" @click="handleReset">重置</el-button>
        </el-form-item>
      </el-form>
    </el-card>

    <!-- 列表 -->
    <el-card shadow="never" class="table-card">
      <div slot="header" class="card-header">
        <div class="card-title">
          <span class="title">商品资料</span>
          <span class="subtitle">共 {{ total }} 条</span>
        </div>
        <div class="card-actions">
          <el-button v-hasPermi="['demo_demo_goods_file:add']" type="primary" icon="el-icon-plus" size="small" @click="openDialog()">新增商品资料</el-button>
          <el-button icon="el-icon-download" size="small" @click="handleExport">导出</el-button>
        </div>
      </div>

      <el-table
        v-loading="loading"
        :data="list"
        class="data-table"
        stripe
        header-cell-class-name="table-head-cell"
        empty-text="暂无数据"
      >
        <el-table-column prop="id" label="ID" width="180" />
        <el-table-column prop="goodsName" label="商品名称" min-width="140" show-overflow-tooltip />
        <el-table-column prop="coverImage" label="商品图" width="140" align="center">
          <template slot-scope="{ row }">
            <el-image v-if="row.coverImage" :src="row.coverImage" :preview-src-list="[row.coverImage]" fit="cover" style="width: 80px; height: 80px; border-radius: 4px">
              <div slot="error"><img src="/placeholder.svg" alt="" style="width: 100%; height: 100%; object-fit: cover" /></div>
            </el-image>
            <span v-else>—</span>
          </template>
        </el-table-column>
        <el-table-column prop="manualFile" label="说明书" min-width="140" show-overflow-tooltip>
          <template slot-scope="{ row }">
            <el-link v-if="row.manualFile" type="primary" :href="row.manualFile" target="_blank" icon="el-icon-document">{{ row.manualFile.split('/').pop() }}</el-link>
          </template>
        </el-table-column>
        <el-table-column prop="updateTime" label="更新时间" width="170" align="center" />
        <el-table-column label="操作" width="278" fixed="right" align="center">
          <template slot-scope="{ row }">
            <el-button type="text" @click="openDetail(row)">详情</el-button>
            <el-button type="text" @click="handlePrintRow(row)">打印</el-button>
            <el-button v-hasPermi="['demo_demo_goods_file:edit']" type="text" @click="openDialog(row)">编辑</el-button>
            <el-button v-hasPermi="['demo_demo_goods_file:remove']" type="text" class="danger-btn" @click="handleDelete(row)">删除</el-button>
          </template>
        </el-table-column>
      </el-table>

      <el-pagination
        class="pagination"
        background
        layout="total, sizes, prev, pager, next, jumper"
        :total="total"
        :page-sizes="[10, 20, 50]"
        :page-size="query.size"
        :current-page.sync="query.page"
        @size-change="handleSizeChange"
        @current-change="fetchList"
      />
    </el-card>

    <!-- 新增/编辑弹窗 -->
    <el-dialog :title="form.id ? '编辑商品资料' : '新增商品资料'" :visible.sync="dialogVisible" width="560px" top="6vh" class="form-dialog">
      <el-form ref="form" :model="form" :rules="rules" label-width="110px">
        <el-form-item label="商品名称" prop="goodsName">
          <el-input v-model="form.goodsName" placeholder="请输入商品名称" />
        </el-form-item>
        <el-form-item label="商品图" prop="coverImage">

      <div slot="footer">
        <el-button @click="dialogVisible = false">取消</el-button>
        <el-button type="primary" :loading="submitting" @click="submitForm">确定</el-button>
      </div>

    <el-drawer title="商品资料详情" :visible.sync="detailVisible" size="420px">

<script>
import {
  getDemoGoodsFileList,
  exportDemoGoodsFile,
  createDemoGoodsFile,
  updateDemoGoodsFile,
  deleteDemoGoodsFile
} from '@/api/goodsFile'
// V6.6 文件能力包：页面里有 image/file 列才引入上传组件（fileEnabled 网控在模型层，未开时列按文本框渲染）
import ImageUpload from '@/components/ImageUpload.vue'
import FileUpload from '@/components/FileUpload.vue'

const defaultForm = () => ({
  id: undefined,
  goodsName: '',
  coverImage: '',
  manualFile: '',
})

export default {
  name: 'DemoGoodsFile',
  components: { ImageUpload, FileUpload },
  data() {
    return {
      // <gen:begin key="vue-data">
      loading: false,
      list: [],
      total: 0,
      query: { page: 1, size: 10, goodsName: '' },
      dialogVisible: false,
      submitting: false,
      detailVisible: false,
      detail: {},
      rules: {
      },
      form: defaultForm()
      // <gen:end key="vue-data">
    }
  },
  created() {
    this.fetchList()
  },
  methods: {
    // <gen:begin key="vue-methods">
    async fetchList() {
      this.loading = true
      try {
        const res = await getDemoGoodsFileList(this.query)
        this.list = res.data.list || []
        this.total = res.data.total || 0
      } catch (e) {
        this.list = []
        this.total = 0
      } finally {
        this.loading = false
      }
    },
    handleSearch() {
      this.query.page = 1
      this.fetchList()
    },
    handleReset() {
      this.query = { page: 1, size: this.query.size, goodsName: '' }
      this.fetchList()
    },
    handleSizeChange(size) {
      this.query.size = size
      this.query.page = 1
      this.fetchList()
    },
    openDetail(row) {
      this.detail = row || {}
      this.detailVisible = true
    },
    openDialog(row) {
      this.form = row ? { ...defaultForm(), ...row } : defaultForm()
      this.dialogVisible = true
      this.$nextTick(() => {
        if (this.$refs.form) this.$refs.form.clearValidate()
      })
    },
    /**
     * 打印单据：跳转独立打印页（无后台导航壳，window.print 直出 A4）
     */
    handlePrintRow(row) {
      this.$router.push('/print/goodsFile/' + row.id)
    },
    async submitForm() {
      const valid = await this.$refs.form.validate().catch(() => false)
      if (!valid) return
      this.submitting = true
      try {
        if (this.form.id) {
          await updateDemoGoodsFile(this.form.id, this.form)
        } else {
          await createDemoGoodsFile(this.form)
        }
        this.$message.success(this.form.id ? '更新成功' : '创建成功')
        this.dialogVisible = false
        this.fetchList()
      } catch (e) { /* 拦截器已提示 */ } finally {
        this.submitting = false
      }
    },
    handleDelete(row) {
      this.$confirm(`确定删除该商品资料吗？`, '提示', { type: 'warning' })
        .then(async () => {
          await deleteDemoGoodsFile(row.id)
          this.$message.success('删除成功')
          if (this.list.length === 1 && this.query.page > 1) this.query.page--
          this.fetchList()
        })
        .catch(() => {})
    },
    async handleExport() {
      try {
        const res = await exportDemoGoodsFile(this.query)
        const blob = new Blob([res.data], { type: 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet' })
        const url = window.URL.createObjectURL(blob)
        const a = document.createElement('a')
        a.href = url
        a.download = '商品资料.xlsx'
        a.click()
        window.URL.revokeObjectURL(url)
      } catch (e) { /* 拦截器已提示 */ }
    }
    // <gen:end key="vue-methods">
    // ===== 以下为手写区：自定义方法写在这里，重新生成不会被覆盖 =====
  }
}
</script>
