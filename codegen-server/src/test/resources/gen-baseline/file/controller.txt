@Operation(summary = "分页查询商品资料")
@PostMapping("/page")
public Result<Page<DemoGoodsFileVO>> page(@RequestBody DemoGoodsFilePageDTO dto) {
    Page<DemoGoodsFile> page = new Page<>(dto.getPageNum(), dto.getPageSize());
    LambdaQueryWrapper<DemoGoodsFile> wrapper = new LambdaQueryWrapper<>();
        wrapper.like(StringUtils.hasText(dto.getGoodsName()), DemoGoodsFile::getGoodsName, dto.getGoodsName());
        wrapper.orderByDesc(DemoGoodsFile::getId);
    demoGoodsFileService.page(page, wrapper);
    Page<DemoGoodsFileVO> voPage = new Page<>();
    BeanUtils.copyProperties(page, voPage);
    List<DemoGoodsFileVO> records = page.getRecords().stream()
            .map(e -> {
                DemoGoodsFileVO vo = new DemoGoodsFileVO();
                BeanUtils.copyProperties(e, vo);
                return vo;
            })
            .toList();
    voPage.setRecords(records);
    return Result.ok(voPage);
}

@Operation(summary = "导出商品资料 Excel（按查询条件过滤；不传条件导出全部）")
@PostMapping("/export")
public ResponseEntity<byte[]> export(@RequestBody(required = false) DemoGoodsFilePageDTO query) throws IOException {
    // 复用列表页的筛选口径：导出的行必须和用户在页面上看到的一致，
    // 否则「筛出 20 条、导出 2000 条」会被当成 bug。
    LambdaQueryWrapper<DemoGoodsFile> wrapper = new LambdaQueryWrapper<>();
    if (query != null) {
        wrapper.like(StringUtils.hasText(query.getGoodsName()), DemoGoodsFile::getGoodsName, query.getGoodsName());
    }
    wrapper.orderByDesc(DemoGoodsFile::getId);
    List<DemoGoodsFile> rows = demoGoodsFileService.list(wrapper);
    try (Workbook wb = new XSSFWorkbook(); ByteArrayOutputStream bos = new ByteArrayOutputStream()) {
        Sheet sheet = wb.createSheet("商品资料");
        Row header = sheet.createRow(0);
        header.createCell(0).setCellValue("商品名称");
        header.createCell(1).setCellValue("商品图");
        header.createCell(2).setCellValue("说明书");
        int rowIndex = 1;
        for (DemoGoodsFile e : rows) {
            Row row = sheet.createRow(rowIndex++);
            row.createCell(0).setCellValue(cellText(e.getGoodsName()));
            row.createCell(1).setCellValue(cellText(e.getCoverImage()));
            row.createCell(2).setCellValue(cellText(e.getManualFile()));
        }
        wb.write(bos);
        String fileName = URLEncoder.encode("商品资料.xlsx", StandardCharsets.UTF_8).replace("+", "%20");
        return ResponseEntity.ok()
                .header(HttpHeaders.CONTENT_DISPOSITION, "attachment; filename*=UTF-8''" + fileName)
                .contentType(MediaType.APPLICATION_OCTET_STREAM)
                .body(bos.toByteArray());
    }
}

@Operation(summary = "根据ID查询商品资料")
@GetMapping("/{id}")
public Result<DemoGoodsFileVO> getById(@PathVariable("id") Long id) {
    DemoGoodsFile entity = demoGoodsFileService.getById(id);
    if (entity == null) {
        return Result.ok(null);
    }
    DemoGoodsFileVO vo = new DemoGoodsFileVO();
    BeanUtils.copyProperties(entity, vo);
    return Result.ok(vo);
}

    demoGoodsFileService.removeById(id);

}

    private static String cellText(Object value) {
        return value == null ? "" : String.valueOf(value);
    }

    private static double cellNum(Object value) {
        return value instanceof Number number ? number.doubleValue() : 0D;
    }

    private static String text(Row row, Integer col) {
        if (col == null || row == null) {
            return null;
        }
        String value = cellText(row.getCell(col)).trim();
        return value.isEmpty() ? null : value;
    }
