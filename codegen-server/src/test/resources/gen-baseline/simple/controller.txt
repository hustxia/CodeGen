@Tag(name = "标签管理", description = "标签管理")
@RestController

public class DemoTagController {

    // Service 注入无条件保留（理由见 import 区注释）

@Operation(summary = "分页查询标签")
@PostMapping("/page")
public Result<Page<DemoTagVO>> page(@RequestBody DemoTagPageDTO dto) {
    Page<DemoTag> page = new Page<>(dto.getPageNum(), dto.getPageSize());
    LambdaQueryWrapper<DemoTag> wrapper = new LambdaQueryWrapper<>();
        wrapper.like(StringUtils.hasText(dto.getTagName()), DemoTag::getTagName, dto.getTagName());
        wrapper.orderByDesc(DemoTag::getId);
    demoTagService.page(page, wrapper);
    Page<DemoTagVO> voPage = new Page<>();
    BeanUtils.copyProperties(page, voPage);
    List<DemoTagVO> records = page.getRecords().stream()
            .map(e -> {
                DemoTagVO vo = new DemoTagVO();
                BeanUtils.copyProperties(e, vo);
                return vo;
            })
            .toList();
    voPage.setRecords(records);
    return Result.ok(voPage);
}

@Operation(summary = "导出标签 Excel（按查询条件过滤；不传条件导出全部）")
@PostMapping("/export")
public ResponseEntity<byte[]> export(@RequestBody(required = false) DemoTagPageDTO query) throws IOException {
    // 复用列表页的筛选口径：导出的行必须和用户在页面上看到的一致，
    // 否则「筛出 20 条、导出 2000 条」会被当成 bug。
    LambdaQueryWrapper<DemoTag> wrapper = new LambdaQueryWrapper<>();
    if (query != null) {
        wrapper.like(StringUtils.hasText(query.getTagName()), DemoTag::getTagName, query.getTagName());
    }
    wrapper.orderByDesc(DemoTag::getId);
    List<DemoTag> rows = demoTagService.list(wrapper);
    try (Workbook wb = new XSSFWorkbook(); ByteArrayOutputStream bos = new ByteArrayOutputStream()) {
        Sheet sheet = wb.createSheet("标签");
        Row header = sheet.createRow(0);
        header.createCell(0).setCellValue("标签名称");
        header.createCell(1).setCellValue("排序");
        int rowIndex = 1;
        for (DemoTag e : rows) {
            Row row = sheet.createRow(rowIndex++);
            row.createCell(0).setCellValue(cellText(e.getTagName()));
            row.createCell(1).setCellValue(cellNum(e.getSortNo()));
        }
        wb.write(bos);
        String fileName = URLEncoder.encode("标签.xlsx", StandardCharsets.UTF_8).replace("+", "%20");
        return ResponseEntity.ok()
                .header(HttpHeaders.CONTENT_DISPOSITION, "attachment; filename*=UTF-8''" + fileName)
                .contentType(MediaType.APPLICATION_OCTET_STREAM)
                .body(bos.toByteArray());
    }
}

@Operation(summary = "根据ID查询标签")
@GetMapping("/{id}")
public Result<DemoTagVO> getById(@PathVariable("id") Long id) {
    DemoTag entity = demoTagService.getById(id);
    if (entity == null) {
        return Result.ok(null);
    }
    DemoTagVO vo = new DemoTagVO();
    BeanUtils.copyProperties(entity, vo);
    return Result.ok(vo);
}

@Operation(summary = "新增标签")
@Log(title = "标签", type = "INSERT")
@PostMapping
public Result<Void> add(@RequestBody DemoTagSaveDTO dto) {
    DemoTag entity = new DemoTag();
    BeanUtils.copyProperties(dto, entity);
    demoTagService.save(entity);
    return Result.ok();
}

@Operation(summary = "修改标签")
@Log(title = "标签", type = "UPDATE")
@PutMapping
public Result<Void> update(@RequestBody DemoTagSaveDTO dto) {
    DemoTag entity = new DemoTag();
    BeanUtils.copyProperties(dto, entity);
    demoTagService.updateById(entity);
    return Result.ok();
}

@Operation(summary = "删除标签")
@Log(title = "标签", type = "DELETE")
@DeleteMapping("/{id}")
public Result<Void> delete(@PathVariable("id") Long id) {
    demoTagService.removeById(id);
    return Result.ok();
}

    private static String cellText(Object value) {
        return value == null ? "" : String.valueOf(value);
    }

    private static double cellNum(Object value) {
        return value instanceof Number number ? number.doubleValue() : 0D;
    }

    private static String text(Row row, Integer col) {
        if (col == null || row == null) {
            return null;
        }
        String value = cellText(row.getCell(col)).trim();
        return value.isEmpty() ? null : value;
    }
