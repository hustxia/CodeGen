@Operation(summary = "分页查询订单")
@PostMapping("/page")
public Result<Page<DemoOrderVO>> page(@RequestBody DemoOrderPageDTO dto) {
    Page<DemoOrder> page = new Page<>(dto.getPageNum(), dto.getPageSize());
    LambdaQueryWrapper<DemoOrder> wrapper = new LambdaQueryWrapper<>();
        wrapper.like(StringUtils.hasText(dto.getOrderNo()), DemoOrder::getOrderNo, dto.getOrderNo());
        wrapper.orderByDesc(DemoOrder::getId);
    demoOrderService.page(page, wrapper);
    Page<DemoOrderVO> voPage = new Page<>();
    BeanUtils.copyProperties(page, voPage);
    List<DemoOrderVO> records = page.getRecords().stream()
            .map(e -> {
                DemoOrderVO vo = new DemoOrderVO();
                BeanUtils.copyProperties(e, vo);
                return vo;
            })
            .toList();
    voPage.setRecords(records);
    return Result.ok(voPage);
}

@Operation(summary = "主表详情（含2张子表明细）")
@GetMapping("/{id}/with-children")
public Result<Map<String, Object>> detailWithChildren(@PathVariable("id") Long id) {
    DemoOrder entity = demoOrderService.getById(id);
    Map<String, Object> data = new LinkedHashMap<>();
    if (entity == null) {
        return Result.ok(data);
    }
    DemoOrderVO masterVo = new DemoOrderVO();
    BeanUtils.copyProperties(entity, masterVo);
    List<Map<String, Object>> children = new ArrayList<>();
    // 子表：订单明细（整棵树一次返回，避免前端逐行发请求）
    Map<String, Object> demoOrderItemGroup = new LinkedHashMap<>();
    demoOrderItemGroup.put("key", "demoOrderItem");
    demoOrderItemGroup.put("tableName", "demo_order_item");
    demoOrderItemGroup.put("comment", "订单明细");
    List<DemoOrderItemVO> demoOrderItemList = new ArrayList<>();
    for (DemoOrderItem row : demoOrderItemService.list(
            new LambdaQueryWrapper<DemoOrderItem>().eq(DemoOrderItem::getOrderId, id))) {
        DemoOrderItemVO childVo = new DemoOrderItemVO();
        BeanUtils.copyProperties(row, childVo);
        // 三级：按子表主键再查一层
        List<DemoOrderItemAttrVO> demoOrderItemAttrList = new ArrayList<>();
        for (DemoOrderItemAttr g : demoOrderItemAttrService.list(
                new LambdaQueryWrapper<DemoOrderItemAttr>()
                        .eq(DemoOrderItemAttr::getItemId, row.getId()))) {
            DemoOrderItemAttrVO gVo = new DemoOrderItemAttrVO();
            BeanUtils.copyProperties(g, gVo);
            demoOrderItemAttrList.add(gVo);
        }
        childVo.setDemoOrderItemAttrs(demoOrderItemAttrList);
        demoOrderItemList.add(childVo);
    }
    demoOrderItemGroup.put("rows", demoOrderItemList);
    children.add(demoOrderItemGroup);
    // 子表：操作日志（整棵树一次返回，避免前端逐行发请求）
    Map<String, Object> demoOrderLogGroup = new LinkedHashMap<>();
    demoOrderLogGroup.put("key", "demoOrderLog");
    demoOrderLogGroup.put("tableName", "demo_order_log");
    demoOrderLogGroup.put("comment", "操作日志");
    List<DemoOrderLogVO> demoOrderLogList = new ArrayList<>();
    for (DemoOrderLog row : demoOrderLogService.list(
            new LambdaQueryWrapper<DemoOrderLog>().eq(DemoOrderLog::getOrderId, id))) {
        DemoOrderLogVO childVo = new DemoOrderLogVO();
        BeanUtils.copyProperties(row, childVo);
        demoOrderLogList.add(childVo);
    }
    demoOrderLogGroup.put("rows", demoOrderLogList);
    children.add(demoOrderLogGroup);
    data.put("master", masterVo);
    data.put("children", children);
    return Result.ok(data);
}

@Operation(summary = "保存订单（主表 + 子表同事务；子表字段为 null 表示不动，空列表表示清空）")
@PostMapping("/save-with-children")
public Result<Void> saveWithChildren(@RequestBody WithChildrenDTO dto) {
    demoOrderService.saveWithChildren(dto.getMaster(),
                                       dto.getDemoOrderItemRows(),
                                       dto.getDemoOrderLogRows());
    return Result.ok();
}

/**
 * 主子表保存入参：每张子表一个可空字段（null=不动该子表，[]=清空该子表）
 */
@lombok.Data
public static class WithChildrenDTO {

    private DemoOrderSaveDTO master;

    private List<DemoOrderItemSaveDTO> demoOrderItemRows;

    private List<DemoOrderLogSaveDTO> demoOrderLogRows;
}

@Operation(summary = "导出订单 Excel（按查询条件过滤；不传条件导出全部）")
@PostMapping("/export")
public ResponseEntity<byte[]> export(@RequestBody(required = false) DemoOrderPageDTO query) throws IOException {
    // 复用列表页的筛选口径：导出的行必须和用户在页面上看到的一致，
    // 否则「筛出 20 条、导出 2000 条」会被当成 bug。
    LambdaQueryWrapper<DemoOrder> wrapper = new LambdaQueryWrapper<>();
    if (query != null) {
        wrapper.like(StringUtils.hasText(query.getOrderNo()), DemoOrder::getOrderNo, query.getOrderNo());
    }
    wrapper.orderByDesc(DemoOrder::getId);
    List<DemoOrder> rows = demoOrderService.list(wrapper);
    try (Workbook wb = new XSSFWorkbook(); ByteArrayOutputStream bos = new ByteArrayOutputStream()) {
        Sheet sheet = wb.createSheet("订单");
        Row header = sheet.createRow(0);
        header.createCell(0).setCellValue("订单号");
        int rowIndex = 1;
        for (DemoOrder e : rows) {
            Row row = sheet.createRow(rowIndex++);
            row.createCell(0).setCellValue(cellText(e.getOrderNo()));
        }
        wb.write(bos);
        String fileName = URLEncoder.encode("订单.xlsx", StandardCharsets.UTF_8).replace("+", "%20");
        return ResponseEntity.ok()
                .header(HttpHeaders.CONTENT_DISPOSITION, "attachment; filename*=UTF-8''" + fileName)
                .contentType(MediaType.APPLICATION_OCTET_STREAM)
                .body(bos.toByteArray());
    }
}

@Operation(summary = "删除订单（按表的级联模式处理明细）")
@Log(title = "订单", type = "DELETE")
@DeleteMapping("/{id}")
public Result<Void> delete(@PathVariable("id") Long id) {

    private static String cellText(Object value) {
        return value == null ? "" : String.valueOf(value);
    }

    private static double cellNum(Object value) {
        return value instanceof Number number ? number.doubleValue() : 0D;
    }

    private static String text(Row row, Integer col) {
        if (col == null || row == null) {
            return null;
        }
        String value = cellText(row.getCell(col)).trim();
        return value.isEmpty() ? null : value;
    }
