    <div class="page-head">
      <div class="page-head__main">
        <h2 class="page-title">请假单</h2>
        <p class="page-desc">维护请假单数据，可按条件筛选、编辑与导出</p>
      </div>
      <div class="page-head__meta">
        <span class="meta-pill">{{ total }} 条记录</span>
      </div>
    </div>
    <!-- 筛选区 -->
    <el-card shadow="never" class="filter-card">
      <el-form :inline="true" :model="query" @submit.native.prevent>
        <el-form-item label="请假事由">
          <el-input v-model="query.reason" placeholder="请假事由" clearable @keyup.enter.native="handleSearch" />
        </el-form-item>
        <el-form-item>
          <el-button v-hasPermi="['demo_leave:query']" type="primary" icon="el-icon-search" @click="handleSearch">查询</el-button>
          <el-button icon="el-icon-refresh" @click="handleReset">重置</el-button>
        </el-form-item>
      </el-form>
    </el-card>

    <!-- 列表 -->
    <el-card shadow="never" class="table-card">
      <div slot="header" class="card-header">
        <div class="card-title">
          <span class="title">请假单</span>
          <span class="subtitle">共 {{ total }} 条</span>
        </div>
        <div class="card-actions">
          <el-button v-hasPermi="['demo_leave:add']" type="primary" icon="el-icon-plus" size="small" @click="openDialog()">新增请假单</el-button>
          <el-button icon="el-icon-download" size="small" @click="handleExport">导出</el-button>
        </div>
      </div>

      <el-table
        v-loading="loading"
        :data="list"
        class="data-table"
        stripe
        header-cell-class-name="table-head-cell"
        empty-text="暂无数据"
      >
        <el-table-column prop="id" label="ID" width="180" />
        <el-table-column prop="reason" label="请假事由" min-width="140" show-overflow-tooltip />
        <el-table-column prop="status" label="状态" min-width="140" show-overflow-tooltip />
        <el-table-column prop="updateTime" label="更新时间" width="170" align="center" />
        <el-table-column label="操作" width="464" fixed="right" align="center">
          <template slot-scope="{ row }">
            <el-button type="text" @click="openDetail(row)">详情</el-button>
            <el-button type="text" @click="handlePrintRow(row)">打印</el-button>
            <!--
              审批按钮按单据状态出现，避免用户对着「已通过」的单子再点提交。
              状态值约定：0 草稿 / 1 待审批 / 2 已通过 / 3 已驳回。
              服务端还会按流程节点配的角色再校验一次，这里只做界面收敛。
            -->
            <el-button v-if="row.status === '0' || row.status === '3'" type="text" @click="handleFlow('submit', row)">提交</el-button>
            <el-button v-if="row.status === '1'" type="text" @click="handleFlow('approve', row)">审批</el-button>
            <el-button v-if="row.status === '1'" type="text" class="danger-btn" @click="handleFlow('reject', row)">驳回</el-button>
            <el-button type="text" @click="openFlowRecords(row)">流程</el-button>
            <el-button v-hasPermi="['demo_leave:edit']" type="text" @click="openDialog(row)">编辑</el-button>
            <el-button v-hasPermi="['demo_leave:remove']" type="text" class="danger-btn" @click="handleDelete(row)">删除</el-button>
          </template>
        </el-table-column>
      </el-table>

      <el-pagination
        class="pagination"
        background
        layout="total, sizes, prev, pager, next, jumper"
        :total="total"
        :page-sizes="[10, 20, 50]"
        :page-size="query.size"
        :current-page.sync="query.page"
        @size-change="handleSizeChange"
        @current-change="fetchList"
      />
    </el-card>

    <!-- 新增/编辑弹窗 -->
    <el-dialog :title="form.id ? '编辑请假单' : '新增请假单'" :visible.sync="dialogVisible" width="560px" top="6vh" class="form-dialog">
      <el-form ref="form" :model="form" :rules="rules" label-width="110px">
        <el-form-item label="请假事由" prop="reason">
          <el-input v-model="form.reason" placeholder="请输入请假事由" />
        </el-form-item>
        <el-form-item label="状态" prop="status">
          <el-input v-model="form.status" placeholder="请输入状态" />
        </el-form-item>
      </el-form>


      <div slot="footer">
        <el-button @click="dialogVisible = false">取消</el-button>
        <el-button type="primary" :loading="submitting" @click="submitForm">确定</el-button>
      </div>
    </el-dialog>

    <!-- 详情抽屉（只读）。主子表：宽抽屉 + el-tabs，tab1 主表字段，tab2..N 每张明细一张只读表；

    <el-drawer title="请假单详情" :visible.sync="detailVisible" size="420px">
      <div class="detail-body">
        <div class="detail-row">
          <span class="detail-label">ID</span>
          <span class="detail-value">{{ detail.id }}</span>
        </div>

        </div>
        <div class="detail-row">
          <span class="detail-label">创建时间</span>
          <span class="detail-value">{{ detail.createTime }}</span>
        </div>
        <div class="detail-row">
          <span class="detail-label">更新时间</span>
          <span class="detail-value">{{ detail.updateTime }}</span>
        </div>
      </div>
    </el-drawer>

    <!-- 提交/审批/驳回的确认框，顺便收一句意见 -->
    <el-dialog
      :title="flowAction === 'submit' ? '提交审批' : (flowAction === 'reject' ? '驳回' : '审批通过')"
      :visible.sync="flowDialogVisible"
      width="420px"
      append-to-body
    >
      <el-input
        v-model="flowComment"
        type="textarea"
        :rows="3"
        maxlength="200"
        show-word-limit
        :placeholder="flowAction === 'reject' ? '请填写驳回理由' : '审批意见（选填）'"
      />
      <div slot="footer">
        <el-button @click="flowDialogVisible = false">取消</el-button>
        <el-button type="primary" :loading="flowSubmitting" @click="submitFlow">确定</el-button>
      </div>
    </el-dialog>
    <!-- 审批记录抽屉：流程流水是只追加的，看的就是「谁在什么时候把单子推到了哪一步」 -->
    <el-drawer title="请假单审批记录" :visible.sync="flowVisible" size="480px">
      <div v-loading="flowLoading" class="detail-body">
        <div v-if="!flowRecords.length" class="flow-empty">还没有流转记录</div>
        <el-timeline v-else>
          <el-timeline-item
            v-for="(r, i) in flowRecords"
            :key="i"
            :timestamp="r.operTime"
            placement="top"
            :color="flowColor(r.action)"
          >
            <div class="flow-node">{{ flowActionText(r.action) }}　<span class="flow-status">{{ flowStatusText(r.toStatus) }}</span></div>
            <div class="flow-meta">{{ r.operName || '系统' }}<span v-if="r.comment">：{{ r.comment }}</span></div>
          </el-timeline-item>
        </el-timeline>

<script>
import {
  getDemoLeaveList,
  exportDemoLeave,
  createDemoLeave,
  updateDemoLeave,
  deleteDemoLeave,
  flowSubmitDemoLeave,
  flowApproveDemoLeave,
  flowRejectDemoLeave,
  getDemoLeaveFlowRecords,
} from '@/api/leave'

const defaultForm = () => ({
  id: undefined,
  reason: '',
  status: '',
})

export default {
  name: 'DemoLeave',
  data() {
    return {
      // <gen:begin key="vue-data">
      loading: false,
      list: [],
      total: 0,
      query: { page: 1, size: 10, reason: '' },
      dialogVisible: false,
      submitting: false,
      detailVisible: false,
      detail: {},
      flowVisible: false,
      flowLoading: false,
      flowRecords: [],
      flowRow: null,
      flowDialogVisible: false,
      flowAction: '',
      flowComment: '',
      flowSubmitting: false,
      rules: {
      },
      form: defaultForm()
      // <gen:end key="vue-data">
    }
  },
  created() {
    this.fetchList()
  },
  methods: {
    // <gen:begin key="vue-methods">
    async fetchList() {
      this.loading = true
      try {
        const res = await getDemoLeaveList(this.query)
        this.list = res.data.list || []
        this.total = res.data.total || 0
      } catch (e) {
        this.list = []
        this.total = 0
      } finally {
        this.loading = false
      }
    },
    handleSearch() {
      this.query.page = 1
      this.fetchList()
    },
    handleReset() {
      this.query = { page: 1, size: this.query.size, reason: '' }
      this.fetchList()
    },
    handleSizeChange(size) {
      this.query.size = size
      this.query.page = 1
      this.fetchList()
    },
    openDetail(row) {
      this.detail = row || {}
      this.detailVisible = true
    },
    openDialog(row) {
      this.form = row ? { ...defaultForm(), ...row } : defaultForm()
      this.dialogVisible = true
      this.$nextTick(() => {
        if (this.$refs.form) this.$refs.form.clearValidate()
      })
    },
    /**
     * 打印单据：跳转独立打印页（无后台导航壳，window.print 直出 A4）
     */
    handlePrintRow(row) {
      this.$router.push('/print/leave/' + row.id)
    },
    async submitForm() {
      const valid = await this.$refs.form.validate().catch(() => false)
      if (!valid) return
      this.submitting = true
      try {
        if (this.form.id) {
          await updateDemoLeave(this.form.id, this.form)
        } else {
          await createDemoLeave(this.form)
        }
        this.$message.success(this.form.id ? '更新成功' : '创建成功')
        this.dialogVisible = false
        this.fetchList()
      } catch (e) { /* 拦截器已提示 */ } finally {
        this.submitting = false
      }
    },
    handleDelete(row) {
      this.$confirm(`确定删除该请假单吗？`, '提示', { type: 'warning' })
        .then(async () => {
          await deleteDemoLeave(row.id)
          this.$message.success('删除成功')
          if (this.list.length === 1 && this.query.page > 1) this.query.page--
          this.fetchList()
        })
        .catch(() => {})
    },
    async handleExport() {
      try {
        const res = await exportDemoLeave(this.query)
        const blob = new Blob([res.data], { type: 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet' })
        const url = window.URL.createObjectURL(blob)
        const a = document.createElement('a')
        a.href = url
        a.download = '请假单.xlsx'
        a.click()
        window.URL.revokeObjectURL(url)
      } catch (e) { /* 拦截器已提示 */ }
    },
    /**
     * 提交 / 审批 / 驳回共用一条通道，避免三份几乎一样的代码。
     * 审批意见走一个小弹窗，不逼用户在列表里用 prompt —— prompt 没法做多行和必填校验。
     */
    handleFlow(action, row) {
      this.flowAction = action
      this.flowRow = row
      this.flowComment = ''
      this.flowDialogVisible = true
    },
    async submitFlow() {
      if (!this.flowRow) return
      const map = {
        submit: { fn: flowSubmitDemoLeave, text: '提交' },
        approve: { fn: flowApproveDemoLeave, text: '审批通过' },
        reject: { fn: flowRejectDemoLeave, text: '驳回' }
      }
      const act = map[this.flowAction]
      if (!act) return
      this.flowSubmitting = true
      try {
        await act.fn(this.flowRow.id, this.flowComment)
        this.$message.success(act.text + '成功')
        this.flowDialogVisible = false
        this.fetchList()
      } catch (e) { /* 拦截器已提示 */ } finally {
        this.flowSubmitting = false
      }
    },
    async openFlowRecords(row) {
      this.flowRow = row
      this.flowRecords = []
      this.flowVisible = true
      this.flowLoading = true
      try {
        const res = await getDemoLeaveFlowRecords(row.id)
        this.flowRecords = res.data || []
      } catch (e) { /* 拦截器已提示 */ } finally {
        this.flowLoading = false
      }
    },
    /** 动作枚举转中文，后端只存 SUBMIT / APPROVE / REJECT */
    flowActionText(action) {
      return ({ SUBMIT: '提交', APPROVE: '审批通过', REJECT: '驳回' })[action] || action
    },
    flowStatusText(status) {
      return ({ '0': '草稿', '1': '待审批', '2': '已通过', '3': '已驳回' })[status] || status
    },
    flowColor(action) {
      return ({ SUBMIT: '#909399', APPROVE: '#67c23a', REJECT: '#f56c6c' })[action] || '#c0c4cc'
    }
    // <gen:end key="vue-methods">
    // ===== 以下为手写区：自定义方法写在这里，重新生成不会被覆盖 =====
  }
}
</script>
