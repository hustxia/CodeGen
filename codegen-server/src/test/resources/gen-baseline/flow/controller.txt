@Operation(summary = "分页查询请假单")
@PostMapping("/page")
public Result<Page<DemoLeaveVO>> page(@RequestBody DemoLeavePageDTO dto) {
    Page<DemoLeave> page = new Page<>(dto.getPageNum(), dto.getPageSize());
    LambdaQueryWrapper<DemoLeave> wrapper = new LambdaQueryWrapper<>();
        wrapper.like(StringUtils.hasText(dto.getReason()), DemoLeave::getReason, dto.getReason());
        wrapper.orderByDesc(DemoLeave::getId);
    demoLeaveService.page(page, wrapper);
    Page<DemoLeaveVO> voPage = new Page<>();
    BeanUtils.copyProperties(page, voPage);
    List<DemoLeaveVO> records = page.getRecords().stream()
            .map(e -> {
                DemoLeaveVO vo = new DemoLeaveVO();
                BeanUtils.copyProperties(e, vo);
                return vo;
            })
            .toList();
    voPage.setRecords(records);
    return Result.ok(voPage);
}

@Operation(summary = "导出请假单 Excel（按查询条件过滤；不传条件导出全部）")
@PostMapping("/export")
public ResponseEntity<byte[]> export(@RequestBody(required = false) DemoLeavePageDTO query) throws IOException {
    // 复用列表页的筛选口径：导出的行必须和用户在页面上看到的一致，
    // 否则「筛出 20 条、导出 2000 条」会被当成 bug。
    LambdaQueryWrapper<DemoLeave> wrapper = new LambdaQueryWrapper<>();
    if (query != null) {
        wrapper.like(StringUtils.hasText(query.getReason()), DemoLeave::getReason, query.getReason());
    }
    wrapper.orderByDesc(DemoLeave::getId);
    List<DemoLeave> rows = demoLeaveService.list(wrapper);
    try (Workbook wb = new XSSFWorkbook(); ByteArrayOutputStream bos = new ByteArrayOutputStream()) {
        Sheet sheet = wb.createSheet("请假单");
        Row header = sheet.createRow(0);
        header.createCell(0).setCellValue("请假事由");
        header.createCell(1).setCellValue("状态");
        int rowIndex = 1;
        for (DemoLeave e : rows) {
            Row row = sheet.createRow(rowIndex++);
            row.createCell(0).setCellValue(cellText(e.getReason()));
            row.createCell(1).setCellValue(cellText(e.getStatus()));
        }
        wb.write(bos);
        String fileName = URLEncoder.encode("请假单.xlsx", StandardCharsets.UTF_8).replace("+", "%20");
        return ResponseEntity.ok()
                .header(HttpHeaders.CONTENT_DISPOSITION, "attachment; filename*=UTF-8''" + fileName)
                .contentType(MediaType.APPLICATION_OCTET_STREAM)
                .body(bos.toByteArray());
    }
}

}

    @Operation(summary = "请假单审批记录")
    @GetMapping("/flow/records/{id}")
    public Result<List<Map<String, Object>>> flowRecords(@PathVariable("id") Long id) {
        List<Map<String, Object>> list = new ArrayList<>();
        for (com.codegen.common.flow.FlowRecord r : flowService.records("hr_leave_flow", id)) {
            Map<String, Object> m = new LinkedHashMap<>();
            m.put("action", r.getAction());
            m.put("fromStatus", r.getFromStatus());
            m.put("toStatus", r.getToStatus());
            m.put("comment", r.getComment());
            m.put("operName", r.getOperName());
            m.put("operTime", r.getOperTime());
            list.add(m);
        }
        return Result.ok(list);
    }

    private static String cellText(Object value) {
        return value == null ? "" : String.valueOf(value);
    }

    private static double cellNum(Object value) {
        return value instanceof Number number ? number.doubleValue() : 0D;
    }

    private static String text(Row row, Integer col) {
        if (col == null || row == null) {
            return null;
        }
        String value = cellText(row.getCell(col)).trim();
        return value.isEmpty() ? null : value;
    }
