import request, { toPageData } from '@/utils/request'

// 缴费单接口：demo_fee（经网关 /api/demo）
// <gen:begin key="api-fns">
function toPageQuery(params) {
  const { page, size, ...query } = params || {}
  return { pageNum: page || 1, pageSize: size || 10, ...query }
}
export function getDemoFeeList(params) {
  return request({
    url: '/api/demo/fee/page',
    method: 'post',
    data: toPageQuery(params)
  }).then(toPageData)
}
export function exportDemoFee(params) {
  return request({
    url: '/api/demo/fee/export',
    method: 'post',
    data: params || {},
    responseType: 'blob'
  })
}
export function getDemoFeeDetail(id) {
  return request({
    url: `/api/demo/fee/${id}`,
    method: 'get'
  })
}
export function createDemoFee(data) {
  return request({
    url: '/api/demo/fee',
    method: 'post',
    data
  })
}
export function updateDemoFee(id, data) {
  return request({
    url: '/api/demo/fee',
    method: 'put',
    data: { ...data, id }
  })
}
export function deleteDemoFee(id) {
  return request({
    url: `/api/demo/fee/${id}`,
    method: 'delete'
  })
}
