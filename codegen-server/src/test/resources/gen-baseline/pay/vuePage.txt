<template>
  <div class="app-container">
    <!-- <gen:begin key="vue-template"> -->
    <!-- 页头：一眼看清「我在哪个页面、这页能干什么」 -->
    <div class="page-head">
      <div class="page-head__main">
        <h2 class="page-title">缴费单</h2>
        <p class="page-desc">维护缴费单数据，可按条件筛选、编辑与导出</p>
      </div>
      <div class="page-head__meta">
        <span class="meta-pill">{{ total }} 条记录</span>
      </div>
    </div>
    <!-- 筛选区 -->
    <el-card shadow="never" class="filter-card">
      <el-form :inline="true" :model="query" @submit.prevent>
        <el-form-item label="缴费项目">
          <el-input v-model="query.feeItem" placeholder="缴费项目" clearable @keyup.enter="handleSearch" />
        </el-form-item>
        <el-form-item>
          <el-button v-hasPermi="['demo_demo_fee:query']" type="primary" icon="Search" @click="handleSearch">查询</el-button>
          <el-button icon="Refresh" @click="handleReset">重置</el-button>
        </el-form-item>
      </el-form>
    </el-card>

    <!-- 列表 -->
    <el-card shadow="never" class="table-card">
      <template #header><div class="card-header">
        <div class="card-title">
          <span class="title">缴费单</span>
          <span class="subtitle">共 {{ total }} 条</span>
        </div>
        <div class="card-actions">
          <el-button v-hasPermi="['demo_demo_fee:add']" type="primary" icon="Plus" size="small" @click="openDialog()">新增缴费单</el-button>
          <el-button icon="Download" size="small" @click="handleExport">导出</el-button>
        </div>
      </div></template>

      <el-table
        v-loading="loading"
        :data="list"
        class="data-table"
        stripe
        header-cell-class-name="table-head-cell"
        empty-text="暂无数据"
      >
        <el-table-column prop="id" label="ID" width="180" />
        <el-table-column prop="feeItem" label="缴费项目" min-width="140" show-overflow-tooltip />
        <el-table-column prop="amount" label="金额" width="110" align="right" show-overflow-tooltip />
        <el-table-column prop="payStatus" label="支付状态" width="110" align="right" show-overflow-tooltip />
        <el-table-column prop="updateTime" label="更新时间" width="170" align="center" />
        <el-table-column label="操作" width="278" fixed="right" align="center">
          <template #default="{ row }">
            <el-button type="text" @click="openDetail(row)">详情</el-button>
            <el-button type="text" @click="handlePrintRow(row)">打印</el-button>
            <!-- V6.3 支付接入约定：待支付的行给「去支付」入口，跳收银台并携带业务定位与金额（amount 列约定为分） -->
            <el-button v-if="row.payStatus === 0 || row.payStatus === '0'" type="text" @click="goPay(row)">去支付</el-button>
            <el-button v-hasPermi="['demo_demo_fee:edit']" type="text" @click="openDialog(row)">编辑</el-button>
            <el-button v-hasPermi="['demo_demo_fee:remove']" type="text" class="danger-btn" @click="handleDelete(row)">删除</el-button>
          </template>
        </el-table-column>
      </el-table>

      <el-pagination
        class="pagination"
        background
        layout="total, sizes, prev, pager, next, jumper"
        :total="total"
        :page-sizes="[10, 20, 50]"
        :page-size="query.size"
        v-model:current-page="query.page"
        @size-change="handleSizeChange"
        @current-change="fetchList"
      />
    </el-card>

    <!-- 新增/编辑弹窗 -->
    <el-dialog :title="form.id ? '编辑缴费单' : '新增缴费单'" v-model="dialogVisible" width="560px" top="6vh" class="form-dialog">
      <el-form ref="form" :model="form" :rules="rules" label-width="110px">
        <el-form-item label="缴费项目" prop="feeItem">
          <el-input v-model="form.feeItem" placeholder="请输入缴费项目" />
        </el-form-item>
        <el-form-item label="金额" prop="amount">
          <el-input v-model="form.amount" placeholder="请输入金额" />
        </el-form-item>
        <el-form-item label="支付状态" prop="payStatus">
          <el-input v-model="form.payStatus" placeholder="请输入支付状态" />
        </el-form-item>
      </el-form>


      <template #footer>
        <el-button @click="dialogVisible = false">取消</el-button>
        <el-button type="primary" :loading="submitting" @click="submitForm">确定</el-button>
      </template>
    </el-dialog>

    <!-- 详情抽屉（只读）。主子表：宽抽屉 + el-tabs，tab1 主表字段，tab2..N 每张明细一张只读表；
         数据复用 /with-children 一次拉全，不逐 tab 发请求 -->
    <el-drawer title="缴费单详情" v-model="detailVisible" size="420px">
      <div class="detail-body">
        <div class="detail-row">
          <span class="detail-label">ID</span>

<script>
import {
  getDemoFeeList,
  exportDemoFee,
  createDemoFee,
  updateDemoFee,
  deleteDemoFee
} from '@/api/fee'

const defaultForm = () => ({
  id: undefined,
  feeItem: '',
  amount: '',
  payStatus: '',
})

export default {
  name: 'DemoFee',
  data() {
    return {
      // <gen:begin key="vue-data">
      loading: false,
      list: [],
      total: 0,
      query: { page: 1, size: 10, feeItem: '' },
      dialogVisible: false,
      submitting: false,
      detailVisible: false,
      detail: {},
      rules: {
      },
      form: defaultForm()
      // <gen:end key="vue-data">
    }
  },
  created() {
    this.fetchList()
  },
  methods: {
    // <gen:begin key="vue-methods">
    async fetchList() {
      this.loading = true
      try {
        const res = await getDemoFeeList(this.query)
        this.list = res.data.list || []
        this.total = res.data.total || 0
      } catch (e) {
        this.list = []
        this.total = 0
      } finally {
        this.loading = false
      }
    },
    handleSearch() {
      this.query.page = 1
      this.fetchList()
    },
    handleReset() {
      this.query = { page: 1, size: this.query.size, feeItem: '' }
      this.fetchList()
    },
    handleSizeChange(size) {
      this.query.size = size
      this.query.page = 1
      this.fetchList()
    },
    openDetail(row) {
      this.detail = row || {}
      this.detailVisible = true
    },
    openDialog(row) {
      this.form = row ? { ...defaultForm(), ...row } : defaultForm()
      this.dialogVisible = true
      this.$nextTick(() => {
        if (this.$refs.form) this.$refs.form.clearValidate()
      })
    },
    /**
     * 打印单据：跳转独立打印页（无后台导航壳，window.print 直出 A4）
     */
    handlePrintRow(row) {
      this.$router.push('/print/fee/' + row.id)
    },
    /**
     * V6.3 去支付：跳收银台，携带业务定位（businessType/businessId）、标题与金额。
     * 金额约定取 amount 列（单位分）；表里没有 amount 列时收银台手动输入。
     */
    goPay(row) {
      this.$router.push({ path: '/pay/cashier', query: {
        subject: '缴费单 ' + row.id,
        amount: row.amount !== undefined && row.amount !== null ? (row.amount / 100).toFixed(2) : '',
        businessType: 'demo_fee',
        businessId: row.id,
        redirect: this.$route.fullPath
      } })
    },
    async submitForm() {
      const valid = await this.$refs.form.validate().catch(() => false)
      if (!valid) return
      this.submitting = true
      try {
        if (this.form.id) {
          await updateDemoFee(this.form.id, this.form)
        } else {
          await createDemoFee(this.form)
        }
        this.$message.success(this.form.id ? '更新成功' : '创建成功')
        this.dialogVisible = false
        this.fetchList()
      } catch (e) { /* 拦截器已提示 */ } finally {
        this.submitting = false
      }
    },
    handleDelete(row) {
      this.$confirm(`确定删除该缴费单吗？`, '提示', { type: 'warning' })
        .then(async () => {
          await deleteDemoFee(row.id)
          this.$message.success('删除成功')
          if (this.list.length === 1 && this.query.page > 1) this.query.page--
          this.fetchList()
        })
        .catch(() => {})
    },
    async handleExport() {
      try {
        const res = await exportDemoFee(this.query)
        const blob = new Blob([res.data], { type: 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet' })
        const url = window.URL.createObjectURL(blob)
        const a = document.createElement('a')
        a.href = url
        a.download = '缴费单.xlsx'
        a.click()
        window.URL.revokeObjectURL(url)
      } catch (e) { /* 拦截器已提示 */ }
    }
    // <gen:end key="vue-methods">
    // ===== 以下为手写区：自定义方法写在这里，重新生成不会被覆盖 =====
  }
}
</script>
