<script>
import {
  getDemoOrderList,
  getDemoOrderWithChildren,
  saveDemoOrderWithChildren,
  exportDemoOrder,
  createDemoOrder,
  updateDemoOrder,
  deleteDemoOrder
} from '@/api/order'

const defaultForm = () => ({
  id: undefined,
  orderNo: '',
  amount: '',
})
// 每张子表的新行工厂：键 = 子表 varName（与后端 children[].key 一致）
const defaultChildMap = {
  'demoOrderItem': () => ({ goodsName: '', qty: '' })
}
// 三级明细新行工厂：键 = '子表.varName'，与 grandDialog 状态对齐
const defaultGrandMap = {
  'demoOrderItem.demoOrderItemAttr': () => ({ attrName: '', attrValue: '' })
}
const grandTitleMap = {
  'demoOrderItem.demoOrderItemAttr': '明细属性'
}

export default {
  name: 'DemoOrder',
  data() {
    return {
      // <gen:begin key="vue-data">
      loading: false,
      list: [],
      total: 0,
      query: { page: 1, size: 10, orderNo: '' },
      dialogVisible: false,
      submitting: false,
      detailVisible: false,
      detail: {},
      // V4.4 详情多 tab：children 组数组 [{key, comment, rows}]，与 /with-children 响应同构
      detailChildren: [],
      activeDetailTab: 'main',
      activeChildTab: 'demoOrderItem',
      demoOrderItemRows: [],
      // 三级明细：grandRows 指向选中行的下级数组（直接改引用，弹窗里编辑等于改主结构），
      // grandDialog 记录当前弹的是哪组（'子表.varName'），决定显示哪张编辑表
      grandVisible: false,
      grandRows: [],
      grandTitle: '',
      grandDialog: '',
      rules: {
      },
      form: defaultForm()
      // <gen:end key="vue-data">
    }
  },
  created() {
    this.fetchList()
  },
  methods: {
    // <gen:begin key="vue-methods">
    async fetchList() {
      this.loading = true
      try {
        const res = await getDemoOrderList(this.query)
        this.list = res.data.list || []
        this.total = res.data.total || 0
      } catch (e) {
        this.list = []
        this.total = 0
      } finally {
        this.loading = false
      }
    },
    handleSearch() {
      this.query.page = 1
      this.fetchList()
    },
    handleReset() {
      this.query = { page: 1, size: this.query.size, orderNo: '' }
      this.fetchList()
    },
    handleSizeChange(size) {
      this.query.size = size
      this.query.page = 1
      this.fetchList()
    },
    openDetail(row) {
      this.detail = row || {}
      this.detailVisible = true
      this.activeDetailTab = 'main'
      this.loadDetailChildren(row && row.id)
    },
    /**
     * 详情抽屉的明细数据：复用 /with-children 一次拉全，不逐 tab 发请求
     */
    async loadDetailChildren(id) {
      this.detailChildren = []
      if (!id) return
      try {
        const res = await getDemoOrderWithChildren(id)
        this.detailChildren = (res.data && res.data.children) || []
      } catch (e) {
        /* 拦截器已提示；明细保持为空 */
      }
    },
    detailGroupRows(key) {
      const group = this.detailChildren.find(g => g.key === key)
      return (group && group.rows) || []
    },
    openDialog(row) {
      this.form = row ? { ...defaultForm(), ...row } : defaultForm()
      this.demoOrderItemRows = []
      this.dialogVisible = true
      if (row && row.id) {
        this.loadChildren(row.id)
      }
      this.$nextTick(() => {
        if (this.$refs.form) this.$refs.form.clearValidate()
      })
    },
    async loadChildren(id) {
      this.demoOrderItemRows = []
      try {
        const res = await getDemoOrderWithChildren(id)
        // 后端 children: [{key, tableName, comment, rows}]；key 即子表 varName
        const groups = (res.data && res.data.children) || []
        groups.forEach(g => {
          if (this[g.key + 'Rows'] !== undefined) {
            this[g.key + 'Rows'] = g.rows || []
          }
        })
      } catch (e) {
        /* 拦截器已提示；保持空明细 */
      }
    },
    addChildRow(tab) {
      const factory = defaultChildMap[tab]
      this[tab + 'Rows'].push(factory ? factory() : {})
    },
    /**
     * 打开某一行的下级明细。用 $set 保障数组存在，
     * 因为后端返回的 VO 里该字段可能为 null（没有下级时）。
     */
    openGrandRows(childVar, gcVar, row) {
      const field = gcVar + 's'
      if (!row[field]) {
        this.$set(row, field, [])
      }
      this.grandRows = row[field]
      this.grandDialog = childVar + '.' + gcVar
      this.grandTitle = grandTitleMap[this.grandDialog] || '下级明细'
      this.grandVisible = true
    },
    addGrandRow() {
      const factory = defaultGrandMap[this.grandDialog]
      this.grandRows.push(factory ? factory() : {})
    },
    /**
     * 打印单据：跳转独立打印页（无后台导航壳，window.print 直出 A4）
     */
    handlePrintRow(row) {
      this.$router.push('/print/order/' + row.id)
    },
    async submitForm() {
      const valid = await this.$refs.form.validate().catch(() => false)
      if (!valid) return
      this.submitting = true
      try {
        // 子表按组提交：与后端约定 null=不动该子表、[]=清空；编辑页全部组都提交（全量替换）
        const payload = { master: this.form }
        payload.demoOrderItemRows = this.demoOrderItemRows
        await saveDemoOrderWithChildren(payload)
        this.$message.success(this.form.id ? '更新成功' : '创建成功')
        this.dialogVisible = false
        this.fetchList()
      } catch (e) { /* 拦截器已提示 */ } finally {
        this.submitting = false
      }
    },
    handleDelete(row) {
      this.$confirm(`确定删除该订单吗？`, '提示', { type: 'warning' })
        .then(async () => {
          await deleteDemoOrder(row.id)
          this.$message.success('删除成功')
          if (this.list.length === 1 && this.query.page > 1) this.query.page--
          this.fetchList()
        })
        .catch(() => {})
    },
    async handleExport() {
      try {
        const res = await exportDemoOrder(this.query)
        const blob = new Blob([res.data], { type: 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet' })
        const url = window.URL.createObjectURL(blob)
        const a = document.createElement('a')
        a.href = url
        a.download = '订单.xlsx'
        a.click()
        window.URL.revokeObjectURL(url)
      } catch (e) { /* 拦截器已提示 */ }
    }
    // <gen:end key="vue-methods">
    // ===== 以下为手写区：自定义方法写在这里，重新生成不会被覆盖 =====
  }
}
</script>
